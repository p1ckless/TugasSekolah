Halaman 90 — Jenis-Jenis Selector CSS

h1 {color: #36CFFF;}

* {color: #36CFFF;}

/* C. Descendant Selector */
ul em {color: #36CFFF;}

.hitam {color: #000000;}

HTML

<!-- 8. Text-transform -->

<p style="text-transform:lowercase;">Semua Huruf Akan Berubah Menjadi Huruf Kecil</p>

<!-- 9. White-space -->
